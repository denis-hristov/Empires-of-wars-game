cmake_minimum_required(VERSION 3.24)
project(EmpiresOfWars VERSION 0.4.7 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EMPIRES_BUILD_CLASSIC "Build the preserved SFML classic desktop edition" OFF)
add_library(kingdom_core src/Config.cpp src/World.cpp src/Game.cpp src/Battle.cpp src/Campaign.cpp src/Laboratory.cpp src/Save.cpp src/Engine.cpp src/Shop.cpp)
target_include_directories(kingdom_core PUBLIC include third_party/json/include)
if(MSVC)
  target_compile_options(kingdom_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(kingdom_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(EMSCRIPTEN)
  target_compile_options(kingdom_core PRIVATE -fexceptions)
  add_executable(empires_engine engine/bindings/WasmBindings.cpp)
  target_link_libraries(empires_engine PRIVATE kingdom_core)
  target_compile_options(empires_engine PRIVATE -fexceptions)
  # Recursive validation of historical saves needs more than the default WASM stack.
  target_link_options(empires_engine PRIVATE --bind -fexceptions -sDISABLE_EXCEPTION_CATCHING=0 -sMODULARIZE=1 -sEXPORT_ES6=1 -sALLOW_MEMORY_GROWTH=1 -sSTACK_SIZE=1048576 -sENVIRONMENT=web,node -sFILESYSTEM=0 -sDYNAMIC_EXECUTION=0)
  set_target_properties(empires_engine PROPERTIES SUFFIX ".js" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/frontend/public/wasm")
else()
  add_library(kingdom_files src/SaveFile.cpp src/ConfigFile.cpp)
  target_link_libraries(kingdom_files PUBLIC kingdom_core)
  enable_testing()
  add_executable(parity_snapshot tests/parity.cpp)
  target_link_libraries(parity_snapshot PRIVATE kingdom_core)
  target_compile_definitions(parity_snapshot PRIVATE EMPIRES_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/game.json")
  add_executable(kingdom_tests tests/tests.cpp)
  target_link_libraries(kingdom_tests PRIVATE kingdom_files)
  target_compile_definitions(kingdom_tests PRIVATE KINGDOM_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/game.json")
  add_test(NAME kingdom_rules COMMAND kingdom_tests)
  add_executable(facade_tests tests/facade.cpp)
  target_link_libraries(facade_tests PRIVATE kingdom_files)
  target_compile_definitions(facade_tests PRIVATE EMPIRES_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME engine_contract COMMAND facade_tests)
  add_executable(military_tests tests/military.cpp)
  target_link_libraries(military_tests PRIVATE kingdom_files)
  target_compile_definitions(military_tests PRIVATE KINGDOM_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/game.json")
  add_test(NAME military_system COMMAND military_tests)
  add_executable(ability_tests tests/abilities.cpp)
  target_link_libraries(ability_tests PRIVATE kingdom_files)
  target_compile_definitions(ability_tests PRIVATE KINGDOM_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/game.json")
  add_test(NAME troop_abilities COMMAND ability_tests)
  add_executable(progression_tests tests/progression.cpp)
  target_link_libraries(progression_tests PRIVATE kingdom_files)
  target_compile_definitions(progression_tests PRIVATE KINGDOM_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/game.json")
  add_test(NAME building_progression COMMAND progression_tests)
  add_executable(economy_tests tests/economy.cpp)
  target_link_libraries(economy_tests PRIVATE kingdom_files)
  target_compile_definitions(economy_tests PRIVATE KINGDOM_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/game.json")
  add_test(NAME production_and_gold COMMAND economy_tests)
  add_executable(unit_footprint_tests tests/unit_footprints.cpp)
  target_link_libraries(unit_footprint_tests PRIVATE kingdom_files)
  target_compile_definitions(unit_footprint_tests PRIVATE KINGDOM_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/game.json")
  add_test(NAME unit_footprints COMMAND unit_footprint_tests)
  add_executable(laboratory_tests tests/laboratory.cpp)
  target_link_libraries(laboratory_tests PRIVATE kingdom_files)
  target_compile_definitions(laboratory_tests PRIVATE KINGDOM_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/game.json")
  add_test(NAME laboratory_potions COMMAND laboratory_tests)
  add_executable(shop_tests tests/shop.cpp)
  target_link_libraries(shop_tests PRIVATE kingdom_files)
  target_compile_definitions(shop_tests PRIVATE KINGDOM_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/game.json")
  add_test(NAME resource_shop COMMAND shop_tests)
  add_executable(advanced_army_tests tests/advanced_army.cpp)
  target_link_libraries(advanced_army_tests PRIVATE kingdom_files)
  target_compile_definitions(advanced_army_tests PRIVATE KINGDOM_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/game.json")
  add_test(NAME advanced_army COMMAND advanced_army_tests)
  add_executable(elite_balance_tests tests/elite_balance.cpp)
  target_link_libraries(elite_balance_tests PRIVATE kingdom_files)
  target_compile_definitions(elite_balance_tests PRIVATE KINGDOM_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/config/game.json")
  add_test(NAME elite_balance COMMAND elite_balance_tests)
  if(EMPIRES_BUILD_CLASSIC)
    include(FetchContent)
    set(SFML_BUILD_AUDIO OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_NETWORK OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SFML_BUILD_TEST_SUITE OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SFML URL https://github.com/SFML/SFML/archive/refs/tags/3.0.2.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(SFML)
    add_executable(EmpiresOfWarsClassic MACOSX_BUNDLE src/main.cpp src/Renderer.cpp)
    target_link_libraries(EmpiresOfWarsClassic PRIVATE kingdom_files SFML::Graphics)
    if(MSVC)
      target_compile_options(EmpiresOfWarsClassic PRIVATE /W4 /utf-8)
    else()
      target_compile_options(EmpiresOfWarsClassic PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    set_target_properties(EmpiresOfWarsClassic PROPERTIES MACOSX_BUNDLE_BUNDLE_NAME "Empires of wars Classic" MACOSX_BUNDLE_GUI_IDENTIFIER "studio.empiresofwars.classic" MACOSX_BUNDLE_BUNDLE_VERSION "0.4.7")
    if(APPLE)
      set(EMPIRES_RESOURCES "$<TARGET_BUNDLE_CONTENT_DIR:EmpiresOfWarsClassic>/Resources")
    else()
      set(EMPIRES_RESOURCES "$<TARGET_FILE_DIR:EmpiresOfWarsClassic>")
    endif()
    add_custom_command(TARGET EmpiresOfWarsClassic POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/config" "${EMPIRES_RESOURCES}/config"
      COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "${EMPIRES_RESOURCES}/assets")
  endif()
endif()
